@MicronautTest
class MeasuredCallInterceptorTest {
    @Inject
    TestService service;

    @Inject
    MeterRegistry registry;
}


@Singleton
class TestService {

    @MeasuredCall(
        value = "test.metric",
        successMetric = "test.metric.success",
        errorMetric = "test.metric.error",
        tags = {"boundness", "test"},
        histogram = true,
        tagPspCode = true
    )
    public String okCall(TestRequest req) {
        return "OK";
    }

    @MeasuredCall(
        value = "test.metric",
        successMetric = "test.metric.success",
        errorMetric = "test.metric.error"
    )
    public String failingCall(TestRequest req) {
        throw new RuntimeException("boom");
    }

    @MeasuredCall(
        value = "test.metric",
        successMetric = "test.metric.success",
        errorMetric = "test.metric.error"
    )
    public CompletionStage<String> asyncCall(TestRequest req) {
        return CompletableFuture.supplyAsync(() -> "ASYNC");
    }
}

class TestRequest {
    private final String pspCode = "ABC123";
    public String getPspCode() { return pspCode; }
}


@MicronautTest
class MeasuredCallInterceptorTest {

    @Inject
    TestService service;

    @Inject
    MeterRegistry registry;

    @Test
    void testSuccessCall_recordsMetrics() {
        service.okCall(new TestRequest());

        assertThat(registry.getMeters().size()).isGreaterThan(0);
    }

    @Test
    void testErrorCall_recordsErrorMetrics() {
        assertThrows(RuntimeException.class, () -> service.failingCall(new TestRequest()));

        assertThat(registry.find("test.metric.error.requests").counter()).isPresent();
    }

    @Test
    void testAsyncCall_recordsMetricsOnCompletion() throws Exception {
        service.asyncCall(new TestRequest()).toCompletableFuture().get(1, TimeUnit.SECONDS);

        assertThat(registry.find("test.metric.success.requests").counter()).isPresent();
    }

    @Test
    void testPspCodeExtractionWorks() {
        service.okCall(new TestRequest());

        Counter c = registry.find("test.metric.success.requests")
                .tag("psp_code", "ABC123")
                .counter()
                .orElseThrow();
        assertThat(c.count()).isGreaterThan(0);
    }

    @Test
    void testContextPropagationAndClear() {
        MetricsContext.push(Tags.of("foo", "bar"), "outerMethod");
        assertThat(MetricsContext.currentTags().iterator().hasNext()).isTrue();
        MetricsContext.clear();
        assertThat(MetricsContext.currentTags().iterator().hasNext()).isFalse();
    }

    @Test
    void testExceptionTagAppearsOnFailure() {
        try {
            service.failingCall(new TestRequest());
        } catch (RuntimeException ignored) {}
        assertThat(registry.find("test.metric.error.requests")
                .tags("exception", "RuntimeException")
                .counter()).isPresent();
    }
}

private Object extractFieldOrAccessor(Object target, String fieldName) {
    if (target == null || fieldName == null) {
        return null;
    }

    Class<?> clazz = target.getClass();

    // Try record accessor first — for records the accessor has the same name as the field
    try {
        Method accessor = clazz.getMethod(fieldName);
        return accessor.invoke(target);
    } catch (NoSuchMethodException ignored) {
        // not a record-style accessor
    } catch (Exception e) {
        return null;
    }

    // Try JavaBean getter style next (getFieldName)
    try {
        String getterName = "get" + Character.toUpperCase(fieldName.charAt(0)) + fieldName.substring(1);
        Method getter = clazz.getMethod(getterName);
        return getter.invoke(target);
    } catch (NoSuchMethodException ignored) {
        // no getter either
    } catch (Exception e) {
        return null;
    }

    // Finally, try to access the field directly
    try {
        Field field = clazz.getDeclaredField(fieldName);
        field.setAccessible(true);
        return field.get(target);
    } catch (Exception e) {
        return null;
    }
}
