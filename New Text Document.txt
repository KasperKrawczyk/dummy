package your.package;

import io.micrometer.core.instrument.*;
import io.micronaut.aop.MethodInvocationContext;
import io.micronaut.core.annotation.AnnotationValue;
import io.micronaut.core.type.Argument;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.mockito.MockedStatic;
import org.mockito.Mockito;

import java.lang.reflect.Field;
import java.lang.reflect.Method;
import java.util.*;

import static org.assertj.core.api.Assertions.*;
import static org.mockito.Mockito.*;

class MeasuredCallInterceptorTest {

    MeterRegistry registry;
    MeasuredCallInterceptor interceptor;

    @BeforeEach
    void setup() {
        registry = mock(MeterRegistry.class);
        interceptor = new MeasuredCallInterceptor(registry);
    }

    // region --- intercept() coverage ---

    @SuppressWarnings("unchecked")
    @Test
    void intercept_success_recordsTimerAndCounter() {
        MethodInvocationContext<Object, Object> ctx = mock(MethodInvocationContext.class);
        Timer.Sample sample = mock(Timer.Sample.class);
        Timer.Builder builder = mock(Timer.Builder.class);
        Timer timer = mock(Timer.class);

        try (MockedStatic<Timer> timerStatic = Mockito.mockStatic(Timer.class)) {
            timerStatic.when(() -> Timer.start(registry)).thenReturn(sample);
            when(Timer.builder(anyString())).thenReturn(builder);
            when(builder.tags(any(Tags.class))).thenReturn(builder);
            when(builder.publishPercentileHistogram(anyBoolean())).thenReturn(builder);
            when(builder.register(registry)).thenReturn(timer);

            AnnotationValue<MeasuredCall> ann = mock(AnnotationValue.class);
            when(ctx.getAnnotation(MeasuredCall.class)).thenReturn(ann);
            when(ctx.getExecutableMethod().getName()).thenReturn("successMethod");
            when(ctx.proceed()).thenReturn("ok");

            Object result = interceptor.intercept(ctx);

            assertThat(result).isEqualTo("ok");
            verify(sample).stop(timer);
            verify(registry, atLeastOnce()).counter(contains("requests"), any(Tags.class));
        }
    }

    @SuppressWarnings("unchecked")
    @Test
    void intercept_exception_recordsErrorTimerAndCounter() {
        MethodInvocationContext<Object, Object> ctx = mock(MethodInvocationContext.class);
        Timer.Sample sample = mock(Timer.Sample.class);
        Timer.Builder builder = mock(Timer.Builder.class);
        Timer timer = mock(Timer.class);
        RuntimeException boom = new RuntimeException("boom");

        try (MockedStatic<Timer> timerStatic = Mockito.mockStatic(Timer.class)) {
            timerStatic.when(() -> Timer.start(registry)).thenReturn(sample);
            when(Timer.builder(anyString())).thenReturn(builder);
            when(builder.tags(any(Tags.class))).thenReturn(builder);
            when(builder.publishPercentileHistogram(anyBoolean())).thenReturn(builder);
            when(builder.register(registry)).thenReturn(timer);

            AnnotationValue<MeasuredCall> ann = mock(AnnotationValue.class);
            when(ctx.getAnnotation(MeasuredCall.class)).thenReturn(ann);
            when(ctx.getExecutableMethod().getName()).thenReturn("errorMethod");
            when(ctx.proceed()).thenThrow(boom);

            assertThatThrownBy(() -> interceptor.intercept(ctx)).isEqualTo(boom);
            verify(sample).stop(timer);
            verify(registry, atLeastOnce()).counter(contains("requests"), any(Tags.class));
        }
    }

    // endregion

    // region --- helper: chooseMetricName() ---
    @Test
    void chooseMetricName_selectsDifferentMetricNames() throws Exception {
        AnnotationValue<MeasuredCall> ann = mock(AnnotationValue.class);
        when(ann.stringValue("successMetric")).thenReturn(Optional.of("successMetricName"));
        when(ann.stringValue("errorMetric")).thenReturn(Optional.of("errorMetricName"));

        Method m = MeasuredCallInterceptor.class.getDeclaredMethod(
                "chooseMetricName", AnnotationValue.class, boolean.class);
        m.setAccessible(true);

        String success = (String) m.invoke(interceptor, ann, true);
        String error = (String) m.invoke(interceptor, ann, false);

        assertThat(success).isEqualTo("successMetricName");
        assertThat(error).isEqualTo("errorMetricName");
    }

    // endregion

    // region --- helper: extractFieldOrAccessor() ---
    @Test
    void extractFieldOrAccessor_fieldAndGetterPaths() throws Exception {
        DummyDto dto = new DummyDto();
        Method m = MeasuredCallInterceptor.class.getDeclaredMethod(
                "extractFieldOrAccessor", Object.class, String.class);
        m.setAccessible(true);

        Object value1 = m.invoke(interceptor, dto, "field");
        Object value2 = m.invoke(interceptor, dto, "getField");
        Object value3 = m.invoke(interceptor, dto, "Field");

        assertThat(value1).isEqualTo("fieldValue");
        assertThat(value2).isEqualTo("fieldValue");
        assertThat(value3).isEqualTo("fieldValue");
    }

    // endregion

    // region --- helper: buildBaseTags() and buildOutcomeTags() ---
    @Test
    void buildBaseTags_withPropagationAndPspCode() throws Exception {
        AnnotationValue<MeasuredCall> ann = mock(AnnotationValue.class);
        MethodInvocationContext<Object, Object> ctx = mock(MethodInvocationContext.class);
        when(ctx.getExecutableMethod().getName()).thenReturn("baseTagMethod");
        when(ctx.getParameterValues()).thenReturn(Map.of("arg0", new Object()));

        when(ann.booleanValue("propagateContext")).thenReturn(Optional.of(true));
        when(ann.booleanValue("addMethodTag")).thenReturn(Optional.of(true));
        when(ann.booleanValue("tagPspCode")).thenReturn(Optional.of(true));
        when(ann.stringValue("pspCodeFieldName")).thenReturn(Optional.of("pspName"));

        Method m = MeasuredCallInterceptor.class.getDeclaredMethod(
                "buildBaseTags", AnnotationValue.class, MethodInvocationContext.class);
        m.setAccessible(true);

        Object result = m.invoke(interceptor, ann, ctx);
        assertThat(result).isInstanceOf(Tags.class);
    }

    @Test
    void buildOutcomeTags_withSuccessAndError() throws Exception {
        AnnotationValue<MeasuredCall> ann = mock(AnnotationValue.class);
        Tags baseTags = Tags.of("base", "value");
        Throwable error = new RuntimeException("boom");

        Method m = MeasuredCallInterceptor.class.getDeclaredMethod(
                "buildOutcomeTags", AnnotationValue.class, Tags.class, boolean.class, Throwable.class);
        m.setAccessible(true);

        Tags successTags = (Tags) m.invoke(interceptor, ann, baseTags, true, null);
        Tags errorTags = (Tags) m.invoke(interceptor, ann, baseTags, false, error);

        assertThat(successTags).isInstanceOf(Tags.class);
        assertThat(errorTags).isInstanceOf(Tags.class);
    }
    // endregion

    // region --- helper: extractPspCode() ---
    @Test
    void extractPspCode_reflectsFromReturnValue() throws Exception {
        MethodInvocationContext<Object, Object> ctx = mock(MethodInvocationContext.class);
        AnnotationValue<MeasuredCall> ann = mock(AnnotationValue.class);
        DummyPsp dto = new DummyPsp();

        when(ctx.proceed()).thenReturn(dto);
        when(ann.stringValue("returnObjectClassSimpleName")).thenReturn(Optional.of("DummyPsp"));
        when(ann.stringValue("pspCodeFieldName")).thenReturn(Optional.of("pspName"));

        Method m = MeasuredCallInterceptor.class.getDeclaredMethod(
                "extractPspCode", MethodInvocationContext.class, AnnotationValue.class);
        m.setAccessible(true);

        String result = (String) m.invoke(interceptor, ctx, ann);
        assertThat(result).isEqualTo("PSP123");
    }
    // endregion

    // region --- private dummy classes for reflection ---
    static class DummyDto {
        private final String field = "fieldValue";
        public String getField() { return field; }
    }

    static class DummyPsp {
        private final String pspName = "PSP123";
        public String getPspName() { return pspName; }
    }
    // endregion
}
